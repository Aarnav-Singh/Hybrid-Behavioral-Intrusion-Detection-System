## 🔹 services/api-gateway/app/main.py

```python
from fastapi import FastAPI, WebSocket
from prometheus_client import Counter, Histogram, generate_latest
from fastapi.responses import Response
import httpx
import time

app = FastAPI()

REQUEST_COUNT = Counter("hb_ids_api_requests_total", "Total API Requests")
REQUEST_LATENCY = Histogram("hb_ids_api_latency_seconds", "API Latency")

DETECTION_URL = "http://detection-service:8001"

@app.get("/api/metrics/system")
def system_metrics():
    start = time.time()
    REQUEST_COUNT.inc()

    try:
        r = httpx.get(f"{DETECTION_URL}/internal/metrics")
        detection_data = r.json()
    except:
        detection_data = {"events_total": 0, "active_alerts": 0}

    REQUEST_LATENCY.observe(time.time() - start)

    return detection_data

@app.websocket("/ws/events")
async def live_events(ws: WebSocket):
    await ws.accept()
    while True:
        await ws.send_json({
            "ip": "192.168.1.15",
            "risk": 0.82,
            "status": "ALERT"
        })
        await asyncio.sleep(2)

@app.get("/metrics")
def metrics():
    return Response(generate_latest(), media_type="text/plain")
```

```python
from fastapi import FastAPI
from elasticsearch import Elasticsearch
from prometheus_client import Counter, Histogram, generate_latest
from fastapi.responses import Response
import time

app = FastAPI()
es = Elasticsearch("http://elasticsearch:9200")

EVENTS_PROCESSED = Counter("hb_ids_events_processed_total", "Total processed events")
INFERENCE_LATENCY = Histogram("hb_ids_inference_latency_seconds", "Model inference latency")

@app.get("/internal/metrics")
def internal_metrics():
    try:
        res = es.count(index="zeek-*")
        total = res["count"]
    except:
        total = 0

    return {
        "events_total": total,
        "active_alerts": 5
    }

@app.get("/metrics")
def metrics():
    return Response(generate_latest(), media_type="text/plain")
```

### JWT Validation (FastAPI auth-service)

```python
from fastapi import FastAPI, Header, HTTPException
import jwt

app = FastAPI()
SECRET = "supersecret"

@app.get("/validate")
def validate(authorization: str = Header(...)):
    token = authorization.replace("Bearer ", "")
    try:
        jwt.decode(token, SECRET, algorithms=["HS256"])
        return {"status": "ok"}
    except:
        raise HTTPException(status_code=401)
```
